### 核心表定义

```sql
-- 科目表
CREATE TABLE subjects (
    id SERIAL PRIMARY KEY,
    name VARCHAR(100) NOT NULL,          -- 科目名称
    code VARCHAR(20) UNIQUE,             -- 科目代码 (如 03708)
    category ENUM('公共课','专业基础课','专业课'),
    total_chapters INT,
    description TEXT
);

-- 知识点表 (树形结构)
CREATE TABLE knowledge_points (
    id SERIAL PRIMARY KEY,
    subject_id INT REFERENCES subjects(id),
    parent_id INT REFERENCES knowledge_points(id),
    name VARCHAR(200),
    level INT,                           -- 层级: 1=章, 2=节, 3=知识点
    frequency_score DECIMAL(5,2),        -- 高频得分 (系统计算)
    weight ENUM('了解','理解','掌握','应用'),
    sort_order INT
);

-- 题目表
CREATE TABLE questions (
    id SERIAL PRIMARY KEY,
    subject_id INT REFERENCES subjects(id),
    knowledge_point_id INT REFERENCES knowledge_points(id),
    exam_year INT,                       -- 出题年份
    exam_month INT,                      -- 出题月份 (4/10)
    question_type ENUM('单选','多选','填空','名词解释','简答','论述','案例分析','计算'),
    difficulty ENUM('易','中','难'),
    content TEXT NOT NULL,               -- 题目内容 (支持富文本/LaTeX)
    options JSONB,                       -- 选项 {"A":"xxx", "B":"xxx", ...}
    answer TEXT,                         -- 标准答案
    explanation TEXT,                    -- 答案解析
    score DECIMAL(4,1),                  -- 分值
    appear_count INT DEFAULT 1,          -- 历年出现次数
    images JSONB,                        -- 关联图片
    created_at TIMESTAMP DEFAULT NOW()
);

-- 视频资源表
CREATE TABLE videos (
    id SERIAL PRIMARY KEY,
    subject_id INT REFERENCES subjects(id),
    knowledge_point_id INT REFERENCES knowledge_points(id),
    title VARCHAR(300),
    platform ENUM('bilibili','netease','tencent','youtube','self'),
    external_url VARCHAR(500),           -- 外部链接
    embed_code TEXT,                     -- 嵌入代码
    duration INT,                        -- 时长(秒)
    view_count INT DEFAULT 0,
    rating DECIMAL(3,2),                 -- 评分
    is_valid BOOLEAN DEFAULT TRUE,       -- 链接是否有效
    last_checked_at TIMESTAMP,
    tags JSONB
);

-- 考试记录表
CREATE TABLE exam_records (
    id SERIAL PRIMARY KEY,
    user_id INT REFERENCES users(id),
    paper_id INT,                        -- 关联试卷
    mode ENUM('真题模拟','随机组卷','章节练习','错题重做'),
    score DECIMAL(5,1),
    total_score DECIMAL(5,1),
    accuracy_rate DECIMAL(5,2),          -- 正确率
    time_spent INT,                      -- 用时(秒)
    answers JSONB,                       -- 作答详情
    started_at TIMESTAMP,
    submitted_at TIMESTAMP
);

-- 错题本
CREATE TABLE wrong_book (
    id SERIAL PRIMARY KEY,
    user_id INT REFERENCES users(id),
    question_id INT REFERENCES questions(id),
    wrong_answer TEXT,
    wrong_count INT DEFAULT 1,           -- 错误次数
    is_mastered BOOLEAN DEFAULT FALSE,   -- 是否已掌握
    note TEXT,                           -- 用户笔记
    last_wrong_at TIMESTAMP,
    created_at TIMESTAMP DEFAULT NOW()
);

-- 学习计划表
CREATE TABLE study_plans (
    id SERIAL PRIMARY KEY,
    user_id INT REFERENCES users(id),
    subject_id INT REFERENCES subjects(id),
    exam_date DATE,                      -- 考试日期
    daily_hours DECIMAL(3,1),            -- 每日学习时长
    plan_data JSONB,                     -- 详细计划 (每日任务)
    progress DECIMAL(5,2) DEFAULT 0,     -- 完成进度 %
    status ENUM('进行中','已完成','已暂停'),
    created_at TIMESTAMP DEFAULT NOW()
);
```

## 🧮 核心算法说明

### 1. 智能组卷算法

```python
"""
基于约束满足的贪心组卷算法
目标: 在满足题型配比、难度配比、章节覆盖的约束下，
     选出最优的题目组合
"""

class PaperGenerator:
    def generate(self, config: PaperConfig) -> Paper:
        """
        config 示例:
        {
            "subject_id": 1,
            "total_score": 100,
            "constraints": {
                "question_types": {"单选": 20, "多选": 10, "简答": 5, "论述": 2},
                "difficulty_ratio": {"易": 0.3, "中": 0.5, "难": 0.2},
                "chapter_coverage": 0.8,  # 至少覆盖80%章节
                "no_repeat_years": 1      # 排除最近1次已做真题
            }
        }
        """
        # Step 1: 构建候选题目池
        candidate_pool = self.query_candidates(config.subject_id)

        # Step 2: 按题型分组，贪心选择
        selected = []
        for q_type, count in config.constraints.question_types.items():
            typed_pool = [q for q in candidate_pool if q.type == q_type]

            # 按难度配比 + 章节多样性 + 去重 贪心选择
            typed_selected = self.greedy_select(
                pool=typed_pool,
                count=count,
                difficulty_ratio=config.constraints.difficulty_ratio,
                exclude_ids=self.get_recent_question_ids(config.no_repeat_years)
            )
            selected.extend(typed_selected)

        # Step 3: 检查章节覆盖率，不足则替换
        coverage = self.calc_chapter_coverage(selected)
        if coverage < config.constraints.chapter_coverage:
            selected = self.optimize_coverage(selected, candidate_pool, config)

        # Step 4: 排序输出 (按题型分区、题号排列)
        return self.build_paper(selected, config)
```

### 2. 高频考点分析算法

```python
"""
考点频次分析与趋势预测
"""

class FrequencyAnalyzer:
    def analyze(self, subject_id: int) -> AnalysisResult:
        # Step 1: 统计各知识点出题频次
        frequency_map = {}
        questions = self.get_all_questions(subject_id)

        for q in questions:
            kp = q.knowledge_point_id
            year = q.exam_year
            frequency_map.setdefault(kp, {"total": 0, "by_year": {}})
            frequency_map[kp]["total"] += 1
            frequency_map[kp]["by_year"].setdefault(year, 0)
            frequency_map[kp]["by_year"][year] += 1

        # Step 2: 计算趋势 (线性回归斜率)
        for kp_id, data in frequency_map.items():
            years = sorted(data["by_year"].keys())
            counts = [data["by_year"].get(y, 0) for y in years]
            data["trend_slope"] = self.linear_regression_slope(years, counts)
            data["trend"] = "上升" if data["trend_slope"] > 0.3 else \
                           "下降" if data["trend_slope"] < -0.3 else "稳定"

        # Step 3: 计算综合权重分 (频次 × 0.6 + 近期趋势 × 0.3 + 分值占比 × 0.1)
        for kp_id, data in frequency_map.items():
            recent_freq = sum(data["by_year"].get(y, 0)
                            for y in range(2022, 2025))
            data["priority_score"] = (
                data["total"] * 0.4 +
                recent_freq * 0.4 +
                max(data["trend_slope"], 0) * 10 * 0.2
            )

        # Step 4: 排序输出
        ranked = sorted(frequency_map.items(),
                       key=lambda x: x[1]["priority_score"], reverse=True)

        return AnalysisResult(
            top_20=ranked[:20],
            trending_up=[kp for kp, d in ranked if d["trend"] == "上升"],
            trending_down=[kp for kp, d in ranked if d["trend"] == "下降"],
            chapter_heatmap=self.build_heatmap(frequency_map),
            wordcloud_data=self.build_wordcloud(frequency_map)
        )
```

### 3. 学习计划生成算法

```python
"""
基于薄弱点分析 + 艾宾浩斯遗忘曲线的学习计划生成
"""

class PlanGenerator:
    # 艾宾浩斯复习间隔 (天)
    REVIEW_INTERVALS = [1, 2, 4, 7, 15, 30]

    def generate(self, user_id: int, config: PlanConfig) -> StudyPlan:
        """
        config: {
            subject_id, exam_date, daily_hours,
            study_preference: "视频优先" | "刷题优先" | "均衡"
        }
        """
        # Step 1: 评估各章节掌握度
        chapter_scores = self.assess_mastery(user_id, config.subject_id)
        # 输出: [{"chapter": "第1章", "mastery": 0.85}, {"chapter": "第3章", "mastery": 0.32}, ...]

        # Step 2: 结合高频考点计算优先级
        frequency_data = FrequencyAnalyzer().analyze(config.subject_id)
        priorities = []
        for ch in chapter_scores:
            freq_weight = self.get_chapter_frequency_weight(ch["chapter"], frequency_data)
            weakness = 1 - ch["mastery"]
            # 优先级 = 薄弱度 × 0.5 + 高频权重 × 0.4 + 分值占比 × 0.1
            priority = weakness * 0.5 + freq_weight * 0.4 + ch["score_ratio"] * 0.1
            priorities.append({
                "chapter": ch["chapter"],
                "priority": priority,
                "mastery": ch["mastery"],
                "estimated_hours": self.estimate_study_hours(weakness, ch["content_volume"])
            })

        priorities.sort(key=lambda x: x["priority"], reverse=True)

        # Step 3: 计算可用时间
        days_remaining = (config.exam_date - date.today()).days
        total_hours = days_remaining * config.daily_hours
        # 预留最后7天做冲刺
        study_days = days_remaining - 7
        sprint_days = 7

        # Step 4: 分配每日任务
        daily_tasks = []
        review_queue = []  # 复习队列 (艾宾浩斯)

        for day in range(study_days):
            task = DailyTask(date=date.today() + timedelta(days=day))
            remaining_hours = config.daily_hours

            # 4a. 先安排复习任务 (艾宾浩斯)
            due_reviews = [r for r in review_queue if r["review_date"] == task.date]
            for review in due_reviews:
                review_time = 0.5  # 每次复习约30分钟
                if remaining_hours >= review_time:
                    task.add_review(review["chapter"], review_time)
                    remaining_hours -= review_time
                    # 安排下一次复习
                    self.schedule_next_review(review, review_queue)

            # 4b. 安排新内容学习
            current_chapter = self.get_current_chapter(priorities, day)
            if current_chapter and remaining_hours > 0:
                # 根据偏好分配: 视频 + 刷题 + 笔记
                video_time, practice_time, note_time = self.allocate_time(
                    remaining_hours, config.study_preference
                )
                task.add_video(current_chapter, video_time)
                task.add_practice(current_chapter, practice_time, question_count=int(practice_time * 20))
                task.add_note(current_chapter, note_time)

                # 加入复习队列 (明天第一次复习)
                review_queue.append({
                    "chapter": current_chapter,
                    "review_date": task.date + timedelta(days=1),
                    "interval_index": 0
                })

            # 4c. 每周末安排一次模考
            if day % 7 == 6:
                task.add_mock_exam()

            daily_tasks.append(task)

        # Step 5: 生成冲刺计划
        sprint_tasks = self.generate_sprint_plan(priorities, sprint_days)
        daily_tasks.extend(sprint_tasks)

        return StudyPlan(
            tasks=daily_tasks,
            total_days=days_remaining,
            priority_chapters=priorities,
            estimated_score_improvement=self.estimate_improvement(priorities)
        )
```
